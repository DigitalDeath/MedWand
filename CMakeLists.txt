cmake_minimum_required(VERSION 3.20)
project(MedicalWand VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MEDWAND_GUI "Build Qt desktop application" ON)
option(MEDWAND_VIEWER "Build viewer as well as Qt persistence" ON)
include(CTest)
add_library(medwand_core src/core.cpp src/reconstruction.cpp src/flat_cap.cpp src/curved_cap.cpp src/radial_reconstruction.cpp)
target_include_directories(medwand_core PUBLIC src)

if(MSVC)
    target_compile_options(medwand_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(medwand_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE medwand_core)
    add_test(NAME core COMMAND core_tests)
    add_executable(cap_tests tests/cap_tests.cpp)
    target_link_libraries(cap_tests PRIVATE medwand_core)
    add_test(NAME flat_cap COMMAND cap_tests)
endif()

if(MEDWAND_GUI)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Test)
    set(CMAKE_AUTOMOC ON)
    add_library(medwand_session src/session_io.cpp)
    target_link_libraries(medwand_session PUBLIC medwand_core Qt6::Core)

    if(MEDWAND_VIEWER)
        add_library(medwand_ui src/viewer.cpp src/main_window.cpp)
        target_link_libraries(medwand_ui PUBLIC medwand_session Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL)
        add_executable(MedicalWand WIN32 src/main.cpp)
        target_link_libraries(MedicalWand PRIVATE medwand_ui)
    endif()

    if(BUILD_TESTING)
        add_executable(session_tests tests/session_tests.cpp)
        target_link_libraries(session_tests PRIVATE medwand_session)
        add_test(NAME session COMMAND session_tests)

        if(MEDWAND_VIEWER)
            add_executable(gui_tests tests/gui_tests.cpp)
            target_link_libraries(gui_tests PRIVATE medwand_ui Qt6::Test)
            add_test(NAME gui COMMAND gui_tests)
            add_test(NAME graphics_failure COMMAND gui_tests --no-graphics)
        endif()
    endif()
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND TARGET MedicalWand)
    install(TARGETS MedicalWand RUNTIME DESTINATION bin)
    install(FILES packaging/medicalwand.desktop DESTINATION share/applications)
    set(CPACK_GENERATOR "DEB")
    set(CPACK_PACKAGE_NAME "medicalwand")
    set(CPACK_PACKAGE_VENDOR "MedWand")
    set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
    set(CPACK_PACKAGE_CONTACT "MedWand project")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "MedicalWand cavity acquisition and surface viewer")
    set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
    set(CPACK_DEBIAN_FILE_NAME "DEB-DEFAULT")
    set(CPACK_DEBIAN_PACKAGE_RELEASE "1")
    set(CPACK_DEBIAN_PACKAGE_SECTION "science")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins, qt6-wayland")
    set(CPACK_DEBIAN_COMPRESSION_TYPE "xz")
    include(CPack)
endif()
